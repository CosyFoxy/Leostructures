
# Build and ignite Fusion Suppressor
giga_contract_build_fusion_suppressor = {
    category = giga_megastructure_contract
    desc = "giga_contract_build_fusion_suppressor_desc"
    event_chain = giga_contract_build_fusion_suppressor_chain
    icon = gfx/interface/icons/situation_log/situation_log_quest.dds
    picture = GFX_evt_supernova
    small_picture = GFX_giga_contract_build_fusion_suppressor_small
    location = yes
    time_to_accept = @contract_time_long
    cost = 2000 #todo decide cost

    lore_issued = giga_contract_issued_lore_build_fusion_suppressor
    lore_completed = giga_contract_completed_lore_build_fusion_suppressor
    lore_failed = giga_contract_failed_lore_build_fusion_suppressor

    potential_issuer = {
        can_issue_contracts_trigger = yes
        contract_system_is_own = yes
        from = {
            any_system_planet = {
                is_colony = no
            }
            has_any_megastructure = no
        }
    }

    potential_operator = {
        NOT = { fromfrom = { is_at_war_with = root } }
        has_technology = giga_tech_fusion_suppressor
    }

    possible_operator = {
        custom_tooltip = {
            fail_text = already_doing_same_contract
            NOT = {
                has_event_chain = giga_contract_build_fusion_suppressor_chain
            }
        }
        contract_target_system_reachable = {
            SYSTEM = from
        }
        custom_tooltip = {
            fail_text = "requires_not_capped"
            OR = {
                has_global_flag = suppressor_capped_u
                check_variable = {
                    which = giga_current_suppressor
                    value < value:giga_suppressor_limit
                }
            }
        }
    }

    issued_abort_trigger = {
        OR = {
            is_nomadic = yes
            fromfrom = {
                NOT = {
                    OR = {
                        any_system_megastructure = { is_upgrading = yes}
                        any_fleet_in_system = { has_fleet_order = build_megastructure_fleet_order }
                        any_system_planet = { is_colony = yes }
                        any_system_planet = { is_under_colonization = yes }
                    }
                }
            }
        }
    }

    abort_trigger = {
        OR = {
            NOT = {
                contract_target_system_reachable = {
                    SYSTEM = fromfrom
                }
            }
            AND = {
                NOT = { has_country_flag = giga_build_fusion_suppressor_contract_succeeded }
                fromfrom = {
                    any_system_megastructure = {
                        giga_is_fusion_suppressor = yes
                    }
                }
            }
        }
    }

    counter = {
        giga_contract_build_fusion_suppressor_counter = {
            max = 1
        }
    }

    ai_weight = {
        weight = 1
    }

    on_monthly = {
        country_event = { id = nomads.6000 }
    }

    on_issue = {
        set_country_flag = giga_build_fusion_suppressor_contract_active
    }

    on_accept = {
        begin_event_chain = {
            event_chain = giga_contract_build_fusion_suppressor_chain
            target = root
        }
        fromfrom = {
            set_star_flag = giga_build_fusion_suppressor_contract_system@this
        }
        create_point_of_interest = {
            id = giga_build_fusion_suppressor_contract_poi@fromfrom
            name = "giga_build_fusion_suppressor_contract_poi_name"
            desc = "giga_build_fusion_suppressor_contract_poi_desc"
            event_chain = giga_contract_build_fusion_suppressor_chain
            location = fromfrom
        }
    }

    on_success = {
        if = {
            limit = {
                exists = root
            }
            #todo balance
            contract_scaling_reward_effect = { TRADE = 1750 RESOURCE = influence RESOURCE_AMOUNT = 35 }
            add_resource = { influence = @opinion_contract_small }
            custom_tooltip = trust15
            custom_tooltip = opinion150
            hidden_effect = {
                prev = { #should be issuer
                    add_opinion_modifier = {
                        modifier = opinion_completed_contract_large
                        who = root
                    }
                    add_trust = {
                        amount = 15
                        who = root
                    }
                }
                fromfrom = {
                    set_star_flag = giga_build_fusion_suppressor_contract_system@this
                    random_system_megastructure = {
                        limit = {
                            giga_is_fusion_suppressor = yes
                        }
                        set_owner = solar_system.owner
                    }
                }
                country_event = { id = nomads.3002 days = 1 }
            }
        }
    }

    on_cancel = {
        if = {
            limit = {
                exists = root
            }
            prev = { #should be issuer
                add_opinion_modifier = {
                    modifier = opinion_failed_contract_large
                    who = root
                }
            }
        }
    }
    on_stop = {
        if = {
            limit = {
                exists = root
            }
            end_event_chain = giga_contract_build_fusion_suppressor_chain
            prev = { remove_country_flag = giga_build_fusion_suppressor_contract_active }
            remove_country_flag = giga_build_fusion_suppressor_contract_succeeded
            # Clear PoI
            remove_point_of_interest = giga_build_fusion_suppressor_contract_poi@fromfrom
        }
    }
}
